// one job, one hex, one vault
#[account] struct Job { hex, poster, worker, amount, state }
 
pub fn fund(ctx, job_id, amount) {
  // poster ATA ─▶ vault PDA (USDC)
  token::transfer(poster → vault, amount)?;
  job.state = Funded;
}
pub fn take(ctx) {
  require!(job.state == Funded && taker != poster);
  job.worker = taker; job.state = Taken;
}
pub fn deliver(ctx) {
  require!(signer == job.worker); job.state = Delivered;
}
pub fn release(ctx) {
  require!(signer == job.poster);
  token::transfer(vault → worker, job.amount)?;
  close(vault); job.state = Paid;
}
pub fn cancel(ctx) { require!(state ≤ Funded); refund(poster) }

Program design shown. The app runs this exact state machine today; on-chain settlement is being wired on Solana devnet.
